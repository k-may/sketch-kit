// Requires three in the host project: npm install three
// Uses the standard sass.txt canvas styles.
import './{sketchname}.scss';
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';
import { BaseSketch } from '../../base/BaseSketch.js';

export default class {sketchname} extends BaseSketch {
    async init() {
        this.scene = new THREE.Scene();
        this.scene.background = new THREE.Color(0x3e3e3e);
        this.scene.fog = new THREE.Fog(0x3e3e3e, 25, 65);
        this.camera = new THREE.PerspectiveCamera(42, 1, 0.1, 100);
        this.camera.position.set(7, 5.5, 8);

        this.renderer = new THREE.WebGLRenderer({ antialias: true });
        this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        this.renderer.shadowMap.enabled = true;
        this.renderer.shadowMap.type = THREE.PCFSoftShadowMap;
        this.renderer.domElement.setAttribute('aria-label', 'Orbiting gray cube on a grid. Drag to orbit and scroll to zoom.');
        this.appendChild(this.renderer.domElement);

        this.scene.add(new THREE.HemisphereLight(0xffffff, 0x555555, 2));
        const light = new THREE.DirectionalLight(0xffffff, 2.5);
        light.position.set(-3, 8, 5);
        light.castShadow = true;
        light.shadow.mapSize.set(2048, 2048);
        Object.assign(light.shadow.camera, { left: -8, right: 8, top: 8, bottom: -8, near: 0.5, far: 30 });
        light.shadow.normalBias = 0.025;
        this.scene.add(light);

        const ground = new THREE.Mesh(
            new THREE.PlaneGeometry(160, 160),
            new THREE.MeshStandardMaterial({ color: 0x414141, roughness: 1 })
        );
        ground.rotation.x = -Math.PI / 2;
        ground.receiveShadow = true;
        this.scene.add(ground);
        const grid = new THREE.GridHelper(100, 100, 0x555555, 0x505050);
        grid.position.y = 0.012;
        this.scene.add(grid);

        // Blender-like red and green world axes on the ground.
        for (const [points, color] of [
            [[[-50, 0.02, 0], [50, 0.02, 0]], 0xb2515c],
            [[[0, 0.02, -50], [0, 0.02, 50]], 0x779d39],
        ]) {
            const geometry = new THREE.BufferGeometry().setFromPoints(points.map(p => new THREE.Vector3(...p)));
            this.scene.add(new THREE.Line(geometry, new THREE.LineBasicMaterial({ color })));
        }

        this.cube = new THREE.Mesh(
            new THREE.BoxGeometry(2, 2, 2),
            new THREE.MeshStandardMaterial({ color: 0x919191, roughness: 0.85 })
        );
        this.cube.position.y = 1;
        this.cube.castShadow = true;
        this.cube.receiveShadow = true;
        this.scene.add(this.cube);
        this.outline = new THREE.LineSegments(
            new THREE.EdgesGeometry(this.cube.geometry),
            new THREE.LineBasicMaterial({ color: 0xffa326 })
        );
        this.outline.scale.setScalar(1.002);
        this.outline.visible = false;
        this.cube.add(this.outline);

        this.controls = new OrbitControls(this.camera, this.renderer.domElement);
        this.controls.target.set(0, 0.85, 0);
        this.controls.enableDamping = true;
        this.controls.autoRotate = true;
        this.controls.autoRotateSpeed = 0.65;
        this.controls.enablePan = false;
        this.controls.minDistance = 4;
        this.controls.maxDistance = 25;
        this.controls.maxPolarAngle = Math.PI / 2 - 0.05;
        this.controls.update();

        this.raycaster = new THREE.Raycaster();
        this.pointer = new THREE.Vector2();
        this.pointerInside = false;
        this.events = new AbortController();
        const options = { signal: this.events.signal };
        this.renderer.domElement.addEventListener('pointermove', event => {
            if (event.pointerType === 'touch') return;
            const rect = this.renderer.domElement.getBoundingClientRect();
            this.pointer.set(
                (event.clientX - rect.left) / rect.width * 2 - 1,
                -(event.clientY - rect.top) / rect.height * 2 + 1
            );
            this.pointerInside = event.clientX >= rect.left && event.clientX <= rect.right &&
                event.clientY >= rect.top && event.clientY <= rect.bottom;
        }, options);
        const clearHover = () => {
            this.pointerInside = false;
            this.outline.visible = false;
        };
        this.renderer.domElement.addEventListener('pointerleave', clearHover, options);
        this.renderer.domElement.addEventListener('pointercancel', clearHover, options);
        window.addEventListener('blur', clearHover, options);
        // The standard stylesheet fixes the canvas, so observe it rather than its parent.
        this.canvasResizeObserver = new ResizeObserver(() => this.onResize());
        this.canvasResizeObserver.observe(this.renderer.domElement);
        // Initialize dimensions before BaseSketch schedules its first draw.
        this.onResize();
    }

    onResize() {
        if (!this.renderer) return;
        const width = this.renderer.domElement.clientWidth || window.innerWidth;
        const height = this.renderer.domElement.clientHeight || window.innerHeight;
        this.camera.aspect = width / Math.max(height, 1);
        this.camera.updateProjectionMatrix();
        this.renderer.setSize(width, height, false);
    }

    draw({ deltaTime }) {
        this.controls.update(Math.min(deltaTime / 1000, 0.05));
        // Recheck every frame: the cube moves under a stationary pointer as we orbit.
        this.scene.updateMatrixWorld(true);
        this.camera.updateMatrixWorld();
        let hovered = false;
        if (this.pointerInside) {
            this.raycaster.setFromCamera(this.pointer, this.camera);
            hovered = this.raycaster.intersectObject(this.cube, false).length > 0;
        }
        this.outline.visible = hovered;
        this.renderer.render(this.scene, this.camera);
    }

    disconnectedCallback() {
        super.disconnectedCallback();
        this.controls?.disconnect();
        this.canvasResizeObserver?.disconnect();
        this.pointerInside = false;
        if (this.outline) this.outline.visible = false;
    }

    // MainView caches and reconnects this instance, including its GPU resources.
    connectedCallback() {
        if (this.controls) this.controls.connect(this.renderer.domElement);
        if (this.canvasResizeObserver) this.canvasResizeObserver.observe(this.renderer.domElement);
        super.connectedCallback();
    }
}
